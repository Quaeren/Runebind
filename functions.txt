    void generateTunnelSegment(Cave& cave, const Cave::StructurePoint& start, const Cave::StructurePoint& end, float shellThickness) {
        float radius = glm::max(start.radius, end.radius);
        glm::vec3 paddingVec(radius + shellThickness + 1.0f);
        
        glm::vec3 minBounds = glm::min(start.position, end.position) - paddingVec;
        glm::vec3 maxBounds = glm::max(start.position, end.position) + paddingVec;

        glm::ivec3 minVoxel = glm::ivec3(glm::floor((minBounds - worldOrigin) / voxelSize));
        glm::ivec3 maxVoxel = glm::ivec3(glm::ceil((maxBounds - worldOrigin) / voxelSize));

        for (int x = minVoxel.x; x <= maxVoxel.x; ++x) {
            for (int y = minVoxel.y; y <= maxVoxel.y; ++y) {
                for (int z = minVoxel.z; z <= maxVoxel.z; ++z) {
                    glm::ivec3 voxelPos(x, y, z);
                    glm::vec3 worldPos = worldOrigin + glm::vec3(voxelPos) * voxelSize;
                    float distance = cave.caveSDF(worldPos);
                    if (distance >= 0.0f && distance <= shellThickness) { addVoxel(voxelPos); }
                }
            }
        }
        
    }
    void generateCavern(Cave& cave, const Cave::Cavern& cavern, float shellThickness) {
        glm::vec3 padding(shellThickness + 1.0f);

        glm::vec3 minBounds = cavern.center - cavern.radii - padding;
        glm::vec3 maxBounds = cavern.center + cavern.radii + padding;
        glm::ivec3 minVoxel = glm::ivec3(glm::floor((minBounds - worldOrigin) / voxelSize));
        glm::ivec3 maxVoxel = glm::ivec3(glm::ceil((maxBounds - worldOrigin) / voxelSize));

        for (int x = minVoxel.x; x <= maxVoxel.x; ++x) {
            for (int y = minVoxel.y; y <= maxVoxel.y; ++y) {
                for (int z = minVoxel.z; z <= maxVoxel.z; ++z) {
                    glm::ivec3 voxelPos(x, y, z);
                    glm::vec3 worldPos = worldOrigin + glm::vec3(voxelPos) * voxelSize;
                    float distance = cave.caveSDF(worldPos);
                    if (distance >= 0.0f && distance <= shellThickness) { addVoxel(voxelPos); }
                }
            }
        }
    }

    void generateFromSDF(Cave& cave, float shellThickness=1.0f) {
        chunks.clear();
        
        const auto& path = cave.getMainPath();
        const auto& sidePaths = cave.getSidePaths();
        const auto& caverns = cave.getCaverns();
        const auto& largeCaverns = cave.getLargeCaverns();

        for (size_t i = 0; i + 1 < path.size(); ++i) {
            generateTunnelSegment(cave, path[i], path[i+1], shellThickness);
        }
        for (const auto& sidePath : sidePaths) {
            for (size_t i = 0; i + 1 < sidePath.size(); ++i) {
                generateTunnelSegment(cave, sidePath[i], sidePath[i+1], shellThickness);
            }
        }
        for (const auto& cavern : caverns) {
            generateCavern(cave, cavern, shellThickness);
        }
        for (const auto& largeCavern : largeCaverns) {
            generateCavern(cave, largeCavern, shellThickness);
        }
    }


    void addVoxelGenerated(const glm::ivec3& pos) {
        glm::ivec3 chunkPos = worldToChunk(pos);
        glm::ivec3 localPos = worldToLocal(pos);

        auto it = chunks.find(chunkPos);
        if (it == chunks.end()) { it = chunks.try_emplace(chunkPos, chunkPos).first; }
        it->second.addVoxel(localPos);
    }

    void generateFromSDF(Cave& cave, float shellThickness=1.0f) {
        chunks.clear();

        std::unordered_set<glm::ivec3, IVec3Hash, IVec3Equal> candidates;
        constexpr float blendStrength = 3.0f;

        auto addBounds = [&](glm::vec3 minBounds, glm::vec3 maxBounds) {
            glm::vec3 padding(shellThickness + blendStrength + voxelSize);
            minBounds -= padding;
            maxBounds += padding;
            glm::vec3 minVoxelFloat = (minBounds - worldOrigin) / voxelSize;
            glm::vec3 maxVoxelFloat = (maxBounds - worldOrigin) / voxelSize;
            glm::ivec3 minVoxel = glm::ivec3(glm::floor(minVoxelFloat));
            glm::ivec3 maxVoxel = glm::ivec3(glm::ceil(maxVoxelFloat));

            for (int x = minVoxel.x; x <= maxVoxel.x; ++x) {
                for (int y = minVoxel.y; y <= maxVoxel.y; ++y) {
                    for (int z = minVoxel.z; z <= maxVoxel.z; ++z) {
                        candidates.emplace(x, y, z);
                    }
                }
            }
        };

        const auto& mainPath = cave.getMainPath();
        for (size_t i = 0; i+1 < mainPath.size(); ++i) {
            const auto& start = mainPath[i];
            const auto& end = mainPath[i+1];
            float radius = glm::max(start.radius, end.radius);
            glm::vec3 padding(radius);

            glm::vec3 minBounds = glm::min(start.position, end.position) - padding;
            glm::vec3 maxBounds = glm::max(start.position, end.position) + padding;
            addBounds(minBounds, maxBounds);
        }
        for (const auto& path : cave.getSidePaths()) {
            for (size_t i = 0; i+1 < path.size(); ++i) {
                const auto& start = path[i];
                const auto& end = path[i+1];
                float radius = glm::max(start.radius, end.radius);
                glm::vec3 padding(radius);

                glm::vec3 minBounds = glm::min(start.position, end.position) - padding;
                glm::vec3 maxBounds = glm::max(start.position, end.position) + padding;
                addBounds(minBounds, maxBounds);
            }
        }

        for (const auto& cavern : cave.getCaverns()) {
            addBounds(cavern.center - cavern.radii, cavern.center + cavern.radii);
        }

        for (const auto& largeCavern : cave.getLargeCaverns()) {
            addBounds(largeCavern.center - largeCavern.radii, largeCavern.center + largeCavern.radii);
            std::cout << "Candidate positions: " << candidates.size() << std::endl;

            for(const auto& voxelPos : candidates) {
                glm::vec3 worldPos = voxelToWorld(voxelPos);
                float distance = cave.caveSDF(worldPos);
                
                if (distance >= 0.0f && distance <= shellThickness) { addVoxelGenerated(voxelPos); }
            }
        }
        std::cout << "Generated Voxels: " << getVoxelCount() << std::endl;
        std::cout << "Generated Chunks: " << getChunkCount() << std::endl;
    }

    float caveSDF(glm::vec3& location, float blendstrength = 3.0f) {
        bool hasShape = false;
        float distance = 0.0f;

        for (size_t i = 0; i + 1 < mainPath.size(); i++) {
            float segmentDistance = tunnelSegmentSDF(location, mainPath[i], mainPath[i + 1]);

            if (!hasShape) {
                distance = segmentDistance;
                hasShape = true;
            } else {
                distance = sMin(distance, segmentDistance, blendstrength);
            }
        }

        for (const auto& sidePath : sidePaths) {
            for (size_t i = 0; i+1 < sidePath.size(); i++) {
                float segmentDistance = tunnelSegmentSDF(location, sidePath[i], sidePath[i + 1]);

                if (!hasShape) {
                    distance = segmentDistance;
                    hasShape = true;
                } else {
                    distance = sMin(distance, segmentDistance, blendstrength);
                }
            }
        }

        for (const auto& cavern : caverns) {
            float cavernDistance = cavernSDF(location, cavern);

            if (!hasShape) {
                distance = cavernDistance;
                hasShape = true;
            } else {
                distance = sMin(distance, cavernDistance, blendstrength);
            }
        }

        for (const auto& largeCavern : largeCaverns) {
            float largeCavernDistance = cavernSDF(location, largeCavern);

            if(!hasShape) {
                distance = largeCavernDistance;
                hasShape = true;
            } else {
                distance = sMin(distance, largeCavernDistance, blendstrength);
            }
        }

        return hasShape ? distance : 100000.0f;
    }


enum CameraMovement {
    FORWARD,
    BACKWARD,
    LEFT,
    RIGHT,
    UP,
    DOWN
};

const float YAW = -90.0f;
const float PITCH = 0.0f;
const float SPEED = 10.0f;
const float SENSITIVITY = 0.1f;
const float FOV = 45.0f;

class Camera {
public:
    // Camera attributes
    glm::vec3 Position;
    glm::vec3 Front;
    glm::vec3 Up;
    glm::vec3 Right;
    glm::vec3 WorldUp;
    // angles
    float Yaw;
    float Pitch;
    // Camera options
    float MovementSpeed;
    float MouseSensitivity;
    float Fov;

    Camera(glm::vec3 position = glm::vec3(0.0f,0.0f,0.0f), glm::vec3 up = glm::vec3(0.0f, 1.0f, 0.0f), float yaw = YAW, float pitch = PITCH) : Front(glm::vec3(0.0f, 0.0f, -1.0f)), MovementSpeed(SPEED), MouseSensitivity(SENSITIVITY), Fov(FOV) {
        Position = position;
        WorldUp = up;
        Yaw = yaw;
        Pitch = pitch;
        updateCameraVectors();
    }

    Camera(float posX, float posY, float posZ, float upX, float upY, float upZ, float yaw, float pitch) : Front(glm::vec3(0.0f, 0.0f, -1.0f)), MovementSpeed(SPEED), MouseSensitivity(SENSITIVITY), Fov(FOV) {
        Position = glm::vec3(posX, posY, posZ);
        WorldUp = glm::vec3(upX, upY, upZ);
        Yaw = yaw;
        Pitch = pitch;
        updateCameraVectors();
    }

    glm::mat4 GetViewMatrix() {
        return glm::lookAt(Position, Position + Front, Up);
    }

    void ProcessKeyboard(CameraMovement direction, float deltaTime) {
        float velocity = MovementSpeed * deltaTime;
        if (direction == FORWARD) {
            Position += Front * velocity;
        }
        if (direction == BACKWARD) {
            Position -= Front * velocity;
        }
        if (direction == LEFT) {
            Position -= Right * velocity;
        }
        if (direction == RIGHT) {
            Position += Right * velocity;
        }
        if (direction == UP) {
            Position += Up * velocity;
        }
        if (direction == DOWN) {
            Position -= Up * velocity;
        }
    }

    void ProcessMouseMovement(float xOffset, float yOffset, GLboolean constrainPitch = true) {
        xOffset *= MouseSensitivity;
        yOffset *= MouseSensitivity;

        Yaw += xOffset;
        Pitch += yOffset;

        if (constrainPitch) {
            if (Pitch > 89.0f) {
                Pitch = 89.0f;
            }
            if (Pitch < -89.0f) {
                Pitch = -89.0f;
            }
        }

        updateCameraVectors();
    }

    void ProcessMouseScroll(float yOffset) {
        Fov -= (float)yOffset;
        if (Fov < 1.0f) {
            Fov = 1.0f;
        }
        if (Fov > 45.0f) {
            Fov = 45.0f;
        }
    }
private:
    
    void updateCameraVectors() {
        glm::vec3 front;
        front.x = cos(glm::radians(Yaw)) * cos(glm::radians(Pitch));
        front.y = sin(glm::radians(Pitch));
        front.z = sin(glm::radians(Yaw)) * cos(glm::radians(Pitch));
        Front = glm::normalize(front);
        
        Right = glm::normalize(glm::cross(Front, WorldUp));
        Up = glm::normalize(glm::cross(Right, Front));
    }
};
